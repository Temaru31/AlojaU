3. CORRER LOCAL (3 terminales - funciona sin BD por mock)
----------------------------------------------------------
# T1 BD (opcional Sprint1, si quieren datos reales)
docker-compose up -d
docker exec -i alojau_db psql -U alojau -d alojau < backend/db/schema.sql
docker exec -i alojau_db psql -U alojau -d alojau < backend/db/seed.sql

# T2 Backend
cd backend
python3 -m venv venv && source venv/bin/activate
pip install -r requirements.txt
cp .env.example .env
uvicorn app.main:app --reload --port 8000  # http://localhost:8000/docs

# T3 Frontend
cd frontend
npm install
cp .env.example .env
npm run dev  # http://localhost:5173
